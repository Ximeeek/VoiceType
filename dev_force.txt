# VoiceType Developer Overrides / Force Settings
#
# Configuration file to force or suppress application behaviors in debug mode.
# This file is completely ignored in production release builds.
#
# Usage:
# Key=Value (e.g., first_start=1)
# 1 = Enabled / Forced
# 0 = Disabled / Suppressed
# Commented out with '#' = Default behavior

# 1. first_start
# Forces the onboarding setup wizard to run on application launch.
# 1 = Force wizard open
# 0 = Suppress wizard open (even if first start is not completed in config)
# Default: Loaded from user configuration (config.toml).
first_start=0

# 2. changelog
# Forces the version update changelog modal to display on launch.
# 1 = Force open changelog modal
# 0 = Suppress open changelog modal
# Default: Shown once on launch when first starting a newly updated version.
changelog=1
